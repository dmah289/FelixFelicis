0 Tổng quan hệ thống

1 Data struct — ParticleRenderData.cs

Mỗi particle = 1 struct 16 bytes. C# layout phải khớp byte-by-byte với HLSL struct.

FieldC# TypeHLSL TypeSize
centerVector2float28B
radiusfloatfloat4B
packedColoruintuint4B
Total16B

16 bytes = power-of-2 → GPU cache line (64/128B) chứa đúng 4/8 particles. Color pack từ 16B (Color) xuống 4B (uint) bằng bit-shift: (R | G<<8 | B<<16 | A<<24).

Tạo file

using System.Runtime.InteropServices;
using UnityEngine;

namespace FelixFelicis.ParticleRendering
{
    [StructLayout(LayoutKind.Sequential)]
    public struct ParticleRenderData
    {
        public Vector2 center;
        public float radius;
        public uint packedColor;

        public static uint PackColor(Color color)
        {
            Color32 c = color;  // implicit clamp [0,1] → [0,255]
            return (uint)(c.r | (c.g << 8) | (c.b << 16) | (c.a << 24));
        }
    }
}
StructLayout(Sequential) ngăn C# compiler sắp xếp lại thứ tự trường. Nếu thứ tự lệch với HLSL → GPU đọc sai offset → visual corruption. Xem ParticleRendering.html §1 để hiểu color packing byte layout chi tiết.

2 Quad mesh — QuadMeshHelper.cs

GPU Instancing vẽ cùng 1 mesh N lần. Mỗi particle dùng 1 quad (4 vertices) — vertex shader scale + position per-instance. Fragment shader dùng SDF quyết định pixel nào thuộc circle.

Tạo file

using UnityEngine;

namespace FelixFelicis.ParticleRendering
{
    public static class QuadMeshHelper
    {
        public static Mesh CreateQuadMesh()
        {
            var mesh = new Mesh { name = "ParticleQuad" };

            mesh.SetVertices(new Vector3[]
            {
                new (-0.5f,  0.5f, 0),  // top-left
                new ( 0.5f,  0.5f, 0),  // top-right
                new (-0.5f, -0.5f, 0),  // bottom-left
                new ( 0.5f, -0.5f, 0),  // bottom-right
            });
            mesh.SetTriangles(new[] { 0, 1, 2, 2, 1, 3 }, 0, true);

            // Giải phóng CPU copy — mesh tĩnh, chỉ GPU cần
            mesh.UploadMeshData(true);

            return mesh;
        }
    }
}
UploadMeshData(true) đánh dấu mesh không thể đọc lại từ CPU — Unity giải phóng managed copy, chỉ giữ GPU-side data.

3 Shader — ParticleDraw + SpaceTransformHelper

Đặt cả 2 file vào folder Resources/ (cùng cấp với Scripts/). Shader load qua Resources.Load<Shader>("ParticleDraw") — guaranteed included trong build.

Resources/SpaceTransformHelper.hlsl

#ifndef SPACE_TRANSFORM_HELPER_INCLUDED
#define SPACE_TRANSFORM_HELPER_INCLUDED

// CBUFFER explicit — bắt buộc cho Vulkan SPIR-V.
// D3D11 tự gom loose globals vào $Globals, Vulkan không có cơ chế đó.
CBUFFER_START(ParticleUniforms)
    float4x4 WorldToClipSpace;   // VP matrix, per-camera
    float2 ScreenSize;           // viewport size
    int useScreenSpace;          // 0 = world, ≠0 = screen
    uint InstanceOffset;         // offset vào StructuredBuffer
CBUFFER_END

float4 WorldToClipPos(float3 worldPos)
{
    if (useScreenSpace)
    {
        float2 uv = worldPos.xy / ScreenSize;
        return float4(uv * 2 - 1, 0, 1);   // NDC-space
    }
    return mul(WorldToClipSpace, float4(worldPos, 1));
}

#endif

Resources/ParticleDraw.shader

Shader "FelixFelicis/ParticleDraw"
{
    SubShader
    {
        Tags
        {
            "RenderType"="Transparent"
            "Queue"="Transparent"
            "RenderPipeline"="UniversalPipeline"  // URP shader stripper cần tag này
        }

        ZWrite Off                          // circles không che nhau theo depth
        ZTest Always                        // overlay trên mọi 3D objects
        Cull Off                            // quad 2D, vẽ cả 2 mặt
        Blend SrcAlpha OneMinusSrcAlpha     // alpha blending chuẩn

        Pass
        {
            HLSLPROGRAM
            #pragma target 4.5
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"
            #include "SpaceTransformHelper.hlsl"

            struct appdata { float4 vertex : POSITION; };

            struct v2f
            {
                float4 clipPos  : SV_POSITION;
                float2 localPos : TEXCOORD0;    // dùng cho SDF
                float  radius   : TEXCOORD1;
                float4 color    : TEXCOORD2;
            };

            struct ParticleData
            {
                float2 center;        // phải khớp C# ParticleRenderData
                float  radius;
                uint   packedColor;
            };

            float4 UnpackColor(uint packed)
            {
                return float4(
                    (packed        & 0xFF) / 255.0,
                    ((packed >> 8)  & 0xFF) / 255.0,
                    ((packed >> 16) & 0xFF) / 255.0,
                    ((packed >> 24) & 0xFF) / 255.0);
            }

            StructuredBuffer<ParticleData> InstanceData;

            v2f vert(appdata v, uint instanceID : SV_InstanceID)
            {
                ParticleData p = InstanceData[instanceID + InstanceOffset];
                v2f o;

                // Quad mở rộng thêm aaPadding cho gradient biên AA
                float2 aaPadding = useScreenSpace ? 2.0 : p.radius * 0.1;
                float2 diameter  = p.radius * 2;
                float2 localVert = v.vertex.xy * (diameter + aaPadding);
                float3 worldPos  = float3(localVert + p.center, 0);

                o.localPos = localVert;
                o.radius   = p.radius;
                o.clipPos  = WorldToClipPos(worldPos);
                o.color    = UnpackColor(p.packedColor);
                return o;
            }

            half4 frag(v2f i) : SV_Target
            {
                // SDF: length(p) - radius. <0 = inside, >0 = outside
                float sdf = length(i.localPos) - i.radius;

                // fwidth = hardware derivative, đo "1 pixel rộng bao nhiêu SDF"
                float fw    = fwidth(sdf);
                // smoothstep Hermite: gradient mượt 1-pixel tại biên
                float alpha = 1.0 - smoothstep(-0.5 * fw, 0.5 * fw, sdf);

                return half4(i.color.rgb, i.color.a * alpha);
            }
            ENDHLSL
        }
    }
}
HLSLPROGRAM, không CGPROGRAM. URP shader stripper strip CGPROGRAM trên Vulkan build. "RenderPipeline"="UniversalPipeline" tag bắt buộc. Xem ParticleRendering.html §3 để hiểu toán SDF, fwidth hardware derivative, smoothstep Hermite curve, và tại sao fwidth thay vì invTexelSize.

4 Core drawer — ParticleDrawer.cs

File lớn nhất — quản lý double-buffered GPU buffers, CPU staging NativeArray, lazy material, draw call. Implement cả IInstanceWriter lẫn IInstanceDrawer.

Kiến thức cần biết trước

ConceptTại sao
Double bufferingCPU và GPU async — 2 buffer luân phiên tránh race condition
NativeArrayUnmanaged heap, không GC — SetData chỉ 1 memcpy
Grow-onlyBuffer không shrink khi count giảm — tránh GPU alloc/dealloc stutter
Resources.LoadShader load bằng path — guaranteed included, khác Shader.Find cần reference
cmd.SetGlobal*()Uniforms vào command stream (deferred) — đúng trên Vulkan, khác material.Set*() (immediate)

Xem ParticleRendering.html §2 & §5 để hiểu sâu NativeArray vs List, 3 cách upload CPU→GPU, double buffer timing, cmd.SetGlobal vs material.Set.

Tạo file

using System;
using System.Runtime.InteropServices;
using Unity.Collections;
using UnityEngine;
using UnityEngine.Rendering;

namespace FelixFelicis.ParticleRendering
{
    public class ParticleDrawer : IInstanceWriter<ParticleRenderData>, IInstanceDrawer, IDisposable
    {
        private static readonly int InstanceDataID   = Shader.PropertyToID("InstanceData");
        private static readonly int InstanceOffsetID = Shader.PropertyToID("InstanceOffset");
        private static readonly int ScreenSizeID     = Shader.PropertyToID("ScreenSize");
        private static readonly int UseScreenSpaceID = Shader.PropertyToID("useScreenSpace");
        private static readonly int WorldToClipID    = Shader.PropertyToID("WorldToClipSpace");

        private static readonly int Stride = Marshal.SizeOf<ParticleRenderData>();

        private const string ShaderResourceName = "ParticleDraw";
        private const int BufferCount = 2;

        private Mesh quadMesh;
        private Material material;
        private readonly uint[] args = new uint[5];

        // Double buffering
        private readonly GraphicsBuffer[] instanceBuffers = new GraphicsBuffer[BufferCount];
        private int writeIndex;
        private GraphicsBuffer argsBuffer;

        // CPU staging — persistent, grow-only
        private NativeArray<ParticleRenderData> cpuStaging;

        private int lastArgsInstanceCount = -1;
        private int currentFrameCount;
        private int lastUploadFrame = -1;
        private bool frameDataReady;

        public ParticleDrawer()
        {
            quadMesh = QuadMeshHelper.CreateQuadMesh();
        }

        // ── Lazy material ──────────────────────────────────────────
        private bool EnsureMaterial()
        {
            if (material != null) return true;
            var shader = Resources.Load<Shader>(ShaderResourceName);
            if (shader == null)
            {
                Debug.LogError($"[ParticleDrawer] Shader '{ShaderResourceName}' not found in Resources.");
                return false;
            }
            material = new Material(shader);
            return true;
        }

        // ── Buffer management ──────────────────────────────────────
        private bool EnsureInstanceBuffer(int index, int count)
        {
            ref var buffer = ref instanceBuffers[index];
            if (buffer != null && buffer.IsValid() && buffer.count >= count)
                return false;
            buffer?.Release();
            buffer = new GraphicsBuffer(GraphicsBuffer.Target.Structured, count, Stride);
            return true;
        }

        private void EnsureStagingCapacity(int count)
        {
            if (cpuStaging.IsCreated && cpuStaging.Length >= count) return;
            if (cpuStaging.IsCreated) cpuStaging.Dispose();
            cpuStaging = new NativeArray<ParticleRenderData>(
                count, Allocator.Persistent, NativeArrayOptions.UninitializedMemory);
        }

        // ── IInstanceWriter ────────────────────────────────────────
        public NativeArray<ParticleRenderData> BeginFrame(int count)
        {
            frameDataReady = false;                          // guard: no stale Draw()
            if (!EnsureMaterial()) return default;
            if (count == 0) return default;
            if (lastUploadFrame == Time.frameCount)          // guard: double call
            {
                Debug.LogWarning("[ParticleDrawer] BeginFrame called twice in the same frame.");
                return default;
            }

            writeIndex = (writeIndex + 1) % BufferCount;     // swap buffer
            EnsureInstanceBuffer(writeIndex, count);
            EnsureStagingCapacity(count);
            currentFrameCount = count;
            return cpuStaging;
        }

        public void EndFrame(int count)
        {
            if (count == 0) return;

            // Upload: 1 memcpy, reliable trên mọi driver kể cả mobile Vulkan
            instanceBuffers[writeIndex].SetData(cpuStaging, 0, 0, count);
            lastUploadFrame = Time.frameCount;
            frameDataReady = true;

            // Args buffer — chỉ re-upload khi count thay đổi
            if (argsBuffer == null || !argsBuffer.IsValid())
            {
                argsBuffer = new GraphicsBuffer(
                    GraphicsBuffer.Target.IndirectArguments, 1, sizeof(uint) * 5);
                args[0] = quadMesh.GetIndexCount(0);   // 6 (2 triangles)
                args[2] = quadMesh.GetIndexStart(0);   // 0
                args[3] = quadMesh.GetBaseVertex(0);   // 0
                args[4] = 0;
                lastArgsInstanceCount = -1;
            }
            if (count != lastArgsInstanceCount)
            {
                args[1] = (uint)count;
                argsBuffer.SetData(args);
                lastArgsInstanceCount = count;
            }
        }

        // ── IInstanceDrawer ────────────────────────────────────────
        public void Draw(CommandBuffer cmd, Camera cam)
        {
            if (!frameDataReady || currentFrameCount == 0) return;

            // Tất cả uniforms qua CommandBuffer — deterministic trên Vulkan
            cmd.SetGlobalBuffer(InstanceDataID, instanceBuffers[writeIndex]);
            cmd.SetGlobalInt(UseScreenSpaceID, 0);
            cmd.SetGlobalInt(InstanceOffsetID, 0);
            cmd.SetGlobalVector(ScreenSizeID,
                new Vector4(cam.pixelWidth, cam.pixelHeight, 0, 0));

            Matrix4x4 vp = GL.GetGPUProjectionMatrix(cam.projectionMatrix, true)
                         * cam.worldToCameraMatrix;
            cmd.SetGlobalMatrix(WorldToClipID, vp);

            cmd.DrawMeshInstancedIndirect(quadMesh, 0, material, 0, argsBuffer);
        }

        // ── IDisposable ────────────────────────────────────────────
        public void Dispose()
        {
            for (int i = 0; i < BufferCount; i++)
            {
                instanceBuffers[i]?.Release();
                instanceBuffers[i] = null;
            }
            argsBuffer?.Release(); argsBuffer = null;

            if (cpuStaging.IsCreated) { cpuStaging.Dispose(); cpuStaging = default; }

            if (material != null)
            {
                if (Application.isPlaying) UnityEngine.Object.Destroy(material);
                else UnityEngine.Object.DestroyImmediate(material);
                material = null;
            }
            if (quadMesh != null)
            {
                if (Application.isPlaying) UnityEngine.Object.Destroy(quadMesh);
                else UnityEngine.Object.DestroyImmediate(quadMesh);
                quadMesh = null;
            }
        }
    }
}

6 Static bridge — ParticleProvider.cs

ScriptableRendererFeature (ScriptableObject) không reference được MonoBehaviour trực tiếp. Cần 1 bridge.

Tạo file

namespace FelixFelicis.ParticleRendering
{
    public static class ParticleProvider
    {
        private static ParticleDrawer instance;

        // Expose qua interface — consumer không biết concrete type
        internal static IInstanceWriter<ParticleRenderData> Writer => instance;
        internal static IInstanceDrawer Drawer => instance;

        internal static void Register(ParticleDrawer drawer) => instance = drawer;

        internal static void Unregister(ParticleDrawer drawer)

            if (instance == drawer) instance = null;  // tránh unregister sai instance
        }
    }

7 URP Feature — ParticleRendererFeature.cs

Composition root — tạo drawer + pass, đăng ký provider, release khi dispose.

Tạo file

using UnityEngine.Rendering.Universal;

namespace FelixFelicis.ParticleRendering
{
    public class ParticleRendererFeature : ScriptableRendererFeature
    {
        private ParticleDrawer drawer;
        private ParticleRenderPass renderPass;

        public override void Create()
        {
            ReleaseResources();   // URP gọi Create() nhiều lần → prevent leak
            drawer = new ParticleDrawer();
            renderPass = new ParticleRenderPass(drawer);
            ParticleProvider.Register(drawer);
        }

        public override void AddRenderPasses(ScriptableRenderer renderer,
            ref RenderingData renderingData)
        {
            if (renderPass != null)
                renderer.EnqueuePass(renderPass);
        }

        protected override void Dispose(bool disposing) => ReleaseResources();

        private void ReleaseResources()
        {
            if (drawer != null)
            {
                ParticleProvider.Unregister(drawer);
                drawer.Dispose();
                drawer = null;
            }
            renderPass = null;
        }
    }
}
ReleaseResources() trong Create(): URP gọi Create() nhiều lần (domain reload, settings change, enter/exit play). Nếu không release trước khi tạo mới → GPU resource leak.

8 URP Pass — ParticleRenderPass.cs

Bridge URP ↔ IInstanceDrawer. RenderGraph path (Unity 6+) + Legacy fallback.

Tạo file

using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.RenderGraphModule;
using UnityEngine.Rendering.Universal;

namespace FelixFelicis.ParticleRendering
{
    public class ParticleRenderPass : ScriptableRenderPass
    {
        private readonly IInstanceDrawer drawer;

        public ParticleRenderPass(IInstanceDrawer drawer)
        {
            this.drawer = drawer;
            renderPassEvent = RenderPassEvent.BeforeRenderingTransparents;
        }

        // ── PassData: class (không struct) — RenderGraph pool instances ──
        private class PassData
        {
            public IInstanceDrawer drawer;
            public Camera camera;
            public TextureHandle colorTarget;
            public TextureHandle depthTarget;
        }

        // ── RenderGraph path ────────────────────────────────────────
        public override void RecordRenderGraph(RenderGraph renderGraph,
            ContextContainer frameData)
        {
            var cameraData   = frameData.Get<UniversalCameraData>();
            var resourceData = frameData.Get<UniversalResourceData>();

            // AddUnsafePass — vì DrawMeshInstancedIndirect là raw cmd op
            using var builder = renderGraph.AddUnsafePass<PassData>(
                "ParticleRendering", out var passData);

            passData.drawer      = drawer;
            passData.camera      = cameraData.camera;
            passData.colorTarget = resourceData.activeColorTexture;
            passData.depthTarget = resourceData.activeDepthTexture;

            builder.UseTexture(resourceData.activeColorTexture, AccessFlags.ReadWrite);
            builder.UseTexture(resourceData.activeDepthTexture, AccessFlags.Read);
            builder.AllowPassCulling(false);  // RenderGraph không detect procedural draw

            // Static lambda → zero closure capture → zero GC
            builder.SetRenderFunc(static (PassData data, UnsafeGraphContext context) =>
            {
                // Explicit bind — AddUnsafePass KHÔNG auto-bind render targets
                context.cmd.SetRenderTarget(data.colorTarget, data.depthTarget);
                var cmd = CommandBufferHelpers.GetNativeCommandBuffer(context.cmd);
                data.drawer.Draw(cmd, data.camera);
            });
        }

        // ── Legacy path ─────────────────────────────────────────────
#pragma warning disable CS0672
        public override void Execute(ScriptableRenderContext context,
            ref RenderingData renderingData)
#pragma warning restore CS0672
        {
            var cmd = CommandBufferPool.Get("ParticleRendering");
            drawer.Draw(cmd, renderingData.cameraData.camera);
            context.ExecuteCommandBuffer(cmd);
            CommandBufferPool.Release(cmd);
        }
    }
}

9 Debug sim — ParticleDebugSim.cs

MonoBehaviour test toàn bộ pipeline. Attach vào bất kỳ GameObject nào. Thay bằng sim thật khi ready.

Tạo file

using Unity.Collections;
using UnityEngine;

namespace FelixFelicis.ParticleRendering
{
    public class ParticleDebugSim : MonoBehaviour
    {
        [Header("Spawn")]
        [SerializeField] private int particleCount = 200;
        [SerializeField] private float spawnRange = 10f;

        [Header("Appearance")]
        [SerializeField] private float radiusMin = 0.1f;
        [SerializeField] private float radiusMax = 0.5f;

        [Header("Animation")]
        [SerializeField] private float driftSpeed = 1f;

        // Pre-allocated — zero per-frame alloc
        private int count;
        private Vector2[] positions;
        private Vector2[] velocities;
        private float[] radii;
        private uint[] packedColors;

        private void Start()
        {
            count = particleCount;
            positions    = new Vector2[count];
            velocities   = new Vector2[count];
            radii        = new float[count];
            packedColors = new uint[count];

            for (int i = 0; i < count; i++)
            {
                positions[i]  = new Vector2(
                    Random.Range(-spawnRange, spawnRange),
                    Random.Range(-spawnRange, spawnRange));
                velocities[i] = Random.insideUnitCircle.normalized;
                radii[i]      = Random.Range(radiusMin, radiusMax);

                // Pack color 1 lần (cold path), không ở Update (hot path)
                packedColors[i] = ParticleRenderData.PackColor(
                    Color.HSVToRGB(Random.value, 0.7f, 1f));
            }
        }

        private void Update()
        {
            float dt = driftSpeed * Time.deltaTime;
            for (int i = 0; i < count; i++)
            {
                positions[i] += velocities[i] * dt;
                if (positions[i].x > spawnRange || positions[i].x < -spawnRange)
                    velocities[i].x = -velocities[i].x;
                if (positions[i].y > spawnRange || positions[i].y < -spawnRange)
                    velocities[i].y = -velocities[i].y;
            }

            // Ghi thẳng vào GPU staging — zero intermediate copy
            var writer = ParticleProvider.Writer;
            if (writer == null) return;

            NativeArray<ParticleRenderData> buffer = writer.BeginFrame(count);
            if (!buffer.IsCreated) return;

            for (int i = 0; i < count; i++)
            {
                buffer[i] = new ParticleRenderData
                {
                    center      = positions[i],
                    radius      = radii[i],
                    packedColor = packedColors[i]
                };
            }
            writer.EndFrame(count);
        }
    }
}

10 Runtime flow — khi nhấn Play

Sau khi tạo xong tất cả files (Step 1–9), thêm Feature vào URP Renderer, attach DebugSim vào GameObject — nhấn Play. Đây là thứ tự chính xác mọi thứ chạy:

Khởi tạo (1 lần)

1. URP load Renderer Asset └── gọi ParticleRendererFeature.Create() ├── ReleaseResources() ← cleanup nếu Create() gọi lại ├── new ParticleDrawer() ← tạo quad mesh ├── new ParticleRenderPass(drawer) └── ParticleProvider.Register(drawer) 2. Unity gọi ParticleDebugSim.Start() ├── pre-allocate arrays (positions, velocities, radii) └── PackColor() cho tất cả particles 1 lần (cold path)

Mỗi frame

3. Unity gọi ParticleDebugSim.Update() ← UPDATE PHASE ├── di chuyển particles (positions += velocities * dt) ├── ParticleProvider.Writer.BeginFrame(count) │ ├── frameDataReady = false │ ├── writeIndex swap (0→1 hoặc 1→0) │ ├── ensure GPU buffer capacity (grow-only) │ ├── ensure CPU staging capacity (grow-only) │ └── return cpuStaging NativeArray ├── ghi vào nativeArray[i] = {center, radius, packedColor} └── Writer.EndFrame(count) ├── SetData(cpuStaging → GPU buffer) ← 1 memcpy ├── update argsBuffer (nếu count đổi) └── frameDataReady = true 4. URP gọi ParticleRendererFeature.AddRenderPasses() ← RENDER PHASE └── enqueue ParticleRenderPass 5. URP execute render pass (per camera) └── ParticleRenderPass.RecordRenderGraph() ├── AddUnsafePass → SetRenderFunc (static lambda): │ ├── context.cmd.SetRenderTarget(color, depth) │ └── drawer.Draw(cmd, cam) │ ├── cmd.SetGlobalBuffer(InstanceData) │ ├── cmd.SetGlobalVector(ScreenSize) │ ├── cmd.SetGlobalMatrix(VP matrix) │ └── DrawMeshInstancedIndirect │ └── GPU vẽ N quads (1 draw call) │ └── fragment shader: SDF → fwidth → smoothstep → circle └── (nếu có camera thứ 2 → Draw() gọi lại, VP matrix mới, data không upload lại)

Quay lại bước 3 cho frame tiếp. writeIndex đã swap → CPU ghi buffer khác, GPU pipeline vẫn đọc buffer cũ an toàn.

Lý thuyết chi tiết: Để hiểu sâu tại sao double buffer, tại sao NativeArray thay vì List, tại sao cmd.SetGlobal thay vì material.Set, tại sao CBUFFER explicit — xem ParticleRendering.html (docs lý thuyết, thư mục Implementation/ParticleRendering/).
